1.) Installation

sudo apt-get install gawk wget git-core diffstat unzip texinfo gcc-multilib \
     build-essential chrpath socat libsdl1.2-dev xterm python3-pip

sudo pip3 install kas

clone this git repository

cd mld-dev
kas build <*.yml>
